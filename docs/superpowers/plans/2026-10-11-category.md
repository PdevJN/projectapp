## Global Constraints

- 全体テスト: `uv run pytest -n auto -q`(直列は `uv run pytest -q`)。型検査: `uvx ty check src`。各タスクの最後に、全体テストが通ること。
- ファイル形式: `FORMAT_VERSION = 3`。古いファイル(キーなしを含む)は既定値で読み(移行を要らなくする)、不正な値は読込を拒否する(`ValueError`)。このアプリより新しい `version` は拒否する(既存の `_check_version`)。
- カテゴリの項目: `name`(最大 `MAX_CATEGORY_NAME_LENGTH` = 30 文字)・`kind`(最大 `MAX_CATEGORY_KIND_LENGTH` = 20 文字)。`name`・`kind` に、改行・タブなどの制御文字(Unicode の `Cc`)は使えない。`prj_code`(最大 `MAX_PROJECT_CODE_LENGTH` = 20 文字)は、半角英数字と `-` `_` `.` だけ(`PRJ_CODE_FORMAT`。空は可)・`expiry_date`(`YYYY-MM-DD` か `None`)・`color`(`blue indigo purple pink red orange amber green teal brown` のどれか)・`prj_code`(最大 `MAX_PROJECT_CODE_LENGTH` = 20 文字)・`id`(8 桁の 16 進。`new_id(taken)`。`compare=False`)。
- タスク名: 最大 `MAX_TASK_NAME_LENGTH` = 50 文字。制御文字(Unicode の `Cc`)は使えない。入力(`build_task`)は拒否し、読込(version を問わない)は拒否せず整える(制御文字は空白に替え、前後の空白を除き、50 文字を超える分は切り詰める)。
- 「未定」: `UNDECIDED = "未定"`。`categories` にも `Project.categories` にも入れない。`category_id is None` が「未定」。名前に「未定」は使えない(登録・読込とも拒否)。
- ガントチャートの見た目は変えない。名前の欄のチップは、カテゴリの `prj_code`(空なら名前)を出し、「未定」のときは出さない。ツールチップはカテゴリ名。チップのマーカー `task-code-<key>-<ti>` と `CODE_CHIP_STYLE` は変えない。
- 書き出し(`todos.json`)は todoapp の形式(`version: 1`)のまま。独自のキーを足さない。`categories` は登録順にすべて(タスクのないものも)書き出す。
- 保存は一時ファイル経由。描画の要素数を増やさない。

def _has_control_char(text: str) -> bool:
    """改行・タブなどの制御文字(Unicode の Cc。C0・DEL・C1)を含むか。"""
    return any(unicodedata.category(ch) == "Cc" for ch in text)


def validate_categories(categories: list[Category]) -> None:
    """カテゴリの一覧を検証する。名前は前後の空白を除いて、空・長すぎ・「未定」・重複を拒否する。名前・種別は制御文字を含むものを、PRJコードは半角英数字と - _ . 以外を含むものを拒否する。"""
    ids: set[str] = set()
    names: set[str] = set()
    for category in categories:
        if not category.id or category.id in ids:
            raise ValueError(f"カテゴリの id が正しくありません: {category.id!r}")
        ids.add(category.id)
        name = category.name.strip()
        if not name:
            raise ValueError("カテゴリの名前を入力してください")
        if name == UNDECIDED:
            raise ValueError(f"「{UNDECIDED}」はカテゴリの名前に使えません")
        if len(name) > MAX_CATEGORY_NAME_LENGTH:
            raise ValueError(f"カテゴリの名前は{MAX_CATEGORY_NAME_LENGTH}文字以内で入力してください")
        if _has_control_char(name):
            raise ValueError("カテゴリの名前に改行・制御文字は使えません")
        if name in names:
            raise ValueError(f"カテゴリの名前が重複しています: {name}")
        names.add(name)
        if category.color not in COLORS:
            raise ValueError(f"カテゴリの色が正しくありません: {category.color!r}")
        if len(category.kind) > MAX_CATEGORY_KIND_LENGTH:
            raise ValueError(f"カテゴリの種別は{MAX_CATEGORY_KIND_LENGTH}文字以内で入力してください")
        if _has_control_char(category.kind):
            raise ValueError("カテゴリの種別に改行・制御文字は使えません")
        if len(category.prj_code) > MAX_PROJECT_CODE_LENGTH:
            raise ValueError(f"PRJコードは{MAX_PROJECT_CODE_LENGTH}文字以内で入力してください")
        if PRJ_CODE_FORMAT.fullmatch(category.prj_code) is None:
            raise ValueError("PRJコードは半角英数字と - _ . だけで入力してください")


def parse_expiry(text: str) -> date | None:
    """有効期限。空は None。YYYY-MM-DD 以外と、年が範囲外のものは ValueError。"""
    value = text.strip()
    if not value:
        return None
    try:
        if not EXPIRY_FORMAT.fullmatch(value):
            raise ValueError
        parsed = date.fromisoformat(value)
    except ValueError:
        raise ValueError(f"有効期限の形式が正しくありません(YYYY-MM-DD): {value}") from None
    if not MIN_YEAR <= parsed.year <= MAX_YEAR:
        raise ValueError(f"有効期限の形式が正しくありません(YYYY-MM-DD): {value}")
    return parsed


def migrate_legacy(pairs: list[tuple[Task, str]], taken: Collection[str] = ()) -> list[Category]:
    """旧 ProjectCode(version 2 以前)をカテゴリへ移す。同じコードごとに 1 つ、出現順に作り、タスクの category_id を書き換える。
    名前はコードそのまま(制御文字は空白に替える)。PRJコードは、使える文字だけでできているときだけ入れる(それ以外は空。チップは名前を出す)。
    空と「未定」(予約名)のコードは、カテゴリを作らず未定のままにする。色は 10 色を出現順に巡回する。"""
    used = set(taken)
    by_name: dict[str, Category] = {}
    for task, code in pairs:
        name = "".join(" " if _has_control_char(ch) else ch for ch in code).strip()
        if not name or name == UNDECIDED:
            continue
        category = by_name.get(name)
        if category is None:
            prj_code = name if PRJ_CODE_FORMAT.fullmatch(name) else ""
            category = Category(name, "", None, COLORS[len(by_name) % len(COLORS)], prj_code, new_id(used))
            used.add(category.id)
            by_name[name] = category
        task.category_id = category.id
    return list(by_name.values())


def next_color(categories: list[Category]) -> str:
    """新しい行の色。使っていない色の先頭。すべて使っていれば、数に応じて巡回する。"""
    used = {c.color for c in categories}
    for color in COLORS:
        if color not in used:
            return color
    return COLORS[len(categories) % len(COLORS)]

def _category_id(value: Any) -> str | None:
    """タスクのカテゴリ id。キーがない・null は未定。空・文字列以外は ValueError(存在の検証は読込の後で行う)。"""
    if value is None:
        return None
    if not isinstance(value, str) or not value:
        raise ValueError(f"タスクのカテゴリの id が正しくありません: {value!r}")
    return value


def _optional_text(value: Any, label: str) -> str:
    """カテゴリの文字列の項目。キーがない・null は空。文字列以外は ValueError。前後の空白を除く。"""
    if value is None:
        return ""
    if not isinstance(value, str):
        raise ValueError(f"{label}が文字列ではありません: {value!r}")
    return value.strip()


def _categories(value: Any) -> list[Category]:
    """カテゴリ(version 3)。キーがない・null は空。名前の空・重複・予約名、色・日付・長さの不正は ValueError。"""
    if value is None:
        return []
    if not isinstance(value, list):
        raise ValueError("カテゴリがリストではありません")
    categories: list[Category] = []
    for raw in value:
        if not isinstance(raw, dict):
            raise ValueError("カテゴリの形式が正しくありません")
        name = raw.get("name")
        if not isinstance(name, str):
            raise ValueError("カテゴリの名前が文字列ではありません")
        expiry = raw.get("expiry_date")
        if expiry is not None and not isinstance(expiry, str):
            raise ValueError(f"カテゴリの有効期限が文字列ではありません: {expiry!r}")
        color = raw.get("color", "blue")
        if color not in COLORS:  # 文字列以外も、ここで落ちる
            raise ValueError(f"カテゴリの色が正しくありません: {color!r}")
        categories.append(
            Category(
                name.strip(),
                _optional_text(raw.get("kind"), "カテゴリの種別"),
                parse_expiry(expiry or ""),
                color,
                _optional_text(raw.get("prj_code"), "PRJコード"),
                _entity_id(raw.get("id"), "カテゴリ"),
            )
        )
    validate_categories(categories)
    return categories

@dataclass
class CategoryRow:
    """設定ダイアログのカテゴリの 1 行。originalは、開いた時点のカテゴリの id(新しい行はNone)。"""

    original: str | None
    name: str
    kind: str
    expiry: str  # YYYY-MM-DD か空
    color: str
    prj_code: str


def build_category_edit(rows: list[CategoryRow], originals: list[Category], taken: Collection[str]) -> CategoryEdit:
    """入力の行を検証して CategoryEdit にする。完全に空の新しい行(色だけ入っている行も)は無視する。新しい行には、既存の id と重ならない id を振る。"""
    used = set(taken)
    built: list[Category] = []
    for row in rows:
        texts = (row.name, row.kind, row.expiry, row.prj_code)
        if row.original is None and not any(t.strip() for t in texts):
            continue
        category_id = row.original or new_id(used)
        used.add(category_id)
        built.append(
            Category(
                row.name.strip(),
                row.kind.strip(),
                parse_expiry(row.expiry),
                row.color,
                row.prj_code.strip(),
                category_id,
            )
        )
    validate_categories(built)
    kept = {c.id for c in built}
    return CategoryEdit(built, [c.id for c in originals if c.id not in kept])
```

def has_control_char(text: str) -> bool:
    """改行・タブなどの制御文字(Unicode の Cc。C0・DEL・C1)を含むか。"""
    return any(unicodedata.category(ch) == "Cc" for ch in text)


def tidy_task_name(name: str) -> str:
    """読込で使う、タスク名の整形。制御文字は空白に替え、前後の空白を除き、上限を超える分は切り詰める(古いファイルを開けなくしない)。"""
    text = "".join(" " if has_control_char(ch) else ch for ch in name).strip()
    return text[:MAX_TASK_NAME_LENGTH].rstrip()
```
