```python
def effort_weight(hours: float) -> float:
    """工数の重み。工数なし・負・有限でない値は 1 時間として数える。"""
    return hours if isfinite(hours) and hours > 0 else 1.0


def effective_progress(state: TaskState) -> float:
    """実効の進捗(0〜100)。終了なら 100、それ以外は進捗度(なければ 0)。"""
    return 100.0 if state.status is Status.DONE else float(state.progress or 0)


def states_at(
    task: Task, events: Sequence[HistoryEvent], moments: Sequence[datetime]
) -> list[TaskState | None]:
    """`state_at`を、昇順の時刻の列にまとめて求める(1 回の走査。日ごとに呼ぶより速い)。
    `events`はこのタスクの分で、日時の昇順。"""
    current = _values(task)
    columns: dict[str, list[Value]] = {}
    for name in HISTORY_FIELDS:
        touching = [
            (e.at, c) for e in events for c in e.changes if c.field == name
        ]
        column: list[Value] = []
        seen = 0
        for moment in moments:
            while seen < len(touching) and touching[seen][0] < moment:
                seen += 1
            if seen:
                column.append(touching[seen - 1][1].after)
            elif touching:
                column.append(touching[0][1].before)
            else:
                column.append(current[name])
        columns[name] = column
    states: list[TaskState | None] = []
    for index in range(len(moments)):
        status = columns["status"][index]
        if status is None:
            states.append(None)
            continue
        progress, effort = columns["progress"][index], columns["effort"][index]
        states.append(
            TaskState(
                Status(str(status)),
                progress if isinstance(progress, int) else None,
                float(effort) if isinstance(effort, (int, float)) else 0.0,
            )
        )
    return states
```

```python
"""残りの工数の推移(要望 128 段階 2)。NiceGUI にも`dashboard`にも依存しない。

`dashboard.summarize_burndown`が、対象のタスクを`BurnTask`にして渡す(循環 import を避けるため)。
"""

from collections.abc import Sequence
from dataclasses import dataclass
from datetime import date, timedelta

from projectapp.models import HistoryEvent, Task
from projectapp.timeline import is_workday


@dataclass(frozen=True)
class BurnTask:
    """推移の対象のタスク 1 件。`start`・`end`は予定の開始・完了予定の日付(読めなければ None)。"""

    task: Task
    events: tuple[HistoryEvent, ...]  # このタスクの履歴(日時の昇順)
    start: date | None
    end: date | None


def count_workdays(first: date, last: date, holidays: dict[date, str]) -> int:
    """`first`〜`last`(両端を含む)の稼働日の数。月〜金で`holidays`の日を除く。"""
    if last < first:
        return 0
    weeks, rest = divmod((last - first).days + 1, 7)
    count = weeks * 5 + sum(
        1 for offset in range(rest) if (first + timedelta(offset)).weekday() < 5
    )
    return count - sum(
        1 for day in holidays if first <= day <= last and day.weekday() < 5
    )


def next_workday(day: date, holidays: dict[date, str]) -> date:
    """その日以降で最初の稼働日。14 日見て無ければその日。"""
    return next(
        (
            d
            for d in (day + timedelta(offset) for offset in range(14))
            if is_workday(d, holidays)
        ),
        day,
    )


def planned_curve(
    tasks: Sequence[BurnTask],
    remains: Sequence[float],
    anchor: date,
    last: date,
    holidays: dict[date, str],
) -> tuple[float, ...]:
    """`anchor`の翌日から`last`までの各日の終わりの、予定の残りの合計。

    タスクごとに、`anchor`時点の残り(`remains`)が、予定の開始(`anchor`の翌日より前なら翌日)から完了予定まで、
    稼働日で直線に下がるとする。完了予定が開始より前(予定超過など)なら開始の 1 日で終える。
    休みの日だけの予定は、次の稼働日に終える扱い。予定が読めないタスクは動かない。
    """
    days = (last - anchor).days
    if days <= 0:
        return ()
    cumulative = [0]  # cumulative[k]: anchor の翌日から k 日目までの稼働日数
    for k in range(1, days + 1):
        cumulative.append(cumulative[-1] + is_workday(anchor + timedelta(k), holidays))
    first_day = anchor + timedelta(1)
    totals = [0.0] * days
    for item, remain in zip(tasks, remains, strict=True):
        if remain <= 0:
            continue
        if item.start is None or item.end is None:
            for index in range(days):
                totals[index] += remain
            continue
        begin = max(item.start, first_day)
        finish = max(item.end, begin)
        if count_workdays(begin, finish, holidays) == 0:
            finish = next_workday(finish, holidays)
        width = max(1, count_workdays(begin, finish, holidays))
        begin_k, finish_k = (begin - anchor).days, (finish - anchor).days
        for k in range(1, days + 1):
            elapsed = (
                0
                if k < begin_k
                else cumulative[min(k, finish_k)] - cumulative[begin_k - 1]
            )
            totals[k - 1] += remain * (1 - min(1.0, elapsed / width))
    return tuple(totals)
```

```python
@dataclass(frozen=True)
class Mark:
    """進捗度が未入力から入った日と、そのタスク名(実際の線の段差の理由)。"""

    day: date
    names: tuple[str, ...]


@dataclass(frozen=True)
class Burndown:
    first: date
    last: date
    known: bool  # 期間の始めまで履歴がある
    since: date | None  # 履歴を記録し始めた日
    count: int = 0  # 対象のタスクの件数
    initial: float = 0.0  # 期間の始めの残り[h]
    actual: tuple[float, ...] = ()  # 日ごとの実際の残り(日の終わり。今日の分は「今」)
    planned: tuple[float, ...] = ()  # 当初の予定の残り(期間の全日。日の終わり)
    future: tuple[float, ...] = ()  # 今後の予定の残り(今日から期間の終わり。先頭は今日の実際)
    workdays: tuple[bool, ...] = ()  # 期間の各日が稼働日か
    today_index: int | None = None  # 今日が期間の何日目か(期間の外は None)
    now_fraction: float = 0.0  # 今日の経過した割合(0〜1)
    marks: tuple[Mark, ...] = ()
    unset: tuple[str, ...] = ()  # 進捗度なしの実行中・一時停止のタスク名
    rest: float | None = None  # 今日の実際 − 期間の終わりの予定の残り(今後の予定の終端)
    required: float | None = None  # 必要なペース[h/稼働日]
    recent: float | None = None  # 直近のペース[h/稼働日]
    recent_days: int = 0  # 直近のペースの窓の稼働日数
    behind: bool = False  # 必要なペースが直近のペースの 1.2 倍を超える
    ahead: bool = False  # 今日の実際が、当初の予定の今日の終わり以下


def _remaining(state: TaskState | None) -> float:
    """タスクの残り。まだ作られていない(None)なら 0。"""
    if state is None:
        return 0.0
    return effort_weight(state.effort) * (1 - effective_progress(state) / 100)


def _by_task(
    tasks: Sequence[BurnTask], moments: Sequence[datetime]
) -> tuple[list[list[float]], list[float]]:
    """タスクごと・時刻ごとの残りと、時刻ごとの合計。"""
    per_task = [
        [_remaining(s) for s in states_at(item.task, item.events, moments)]
        for item in tasks
    ]
    totals = [sum(row[i] for row in per_task) for i in range(len(moments))]
    return per_task, totals


def _recent_pace(
    tasks: Sequence[BurnTask],
    today: date,
    now: datetime,
    since: datetime,
    holidays: dict[date, str],
) -> tuple[float | None, int]:
    """直近 10 稼働日の正味のペース[h/稼働日]。窓の始めは記録開始以降に縮める。3 稼働日に満たなければ出さない。"""
    day, found, window_start = today - timedelta(1), 0, None
    for _ in range(366):
        if found >= WINDOW_DAYS or datetime.combine(day, time.min) < since:
            break
        window_start = day
        found += is_workday(day, holidays)
        day -= timedelta(1)
    if window_start is None:
        return None, 0
    days = count_workdays(window_start, today - timedelta(1), holidays)
    if days < MIN_WINDOW_DAYS:
        return None, 0
    _, totals = _by_task(tasks, [datetime.combine(window_start, time.min), now])
    return max(0.0, (totals[0] - totals[1]) / days), days


def _marks(
    tasks: Sequence[BurnTask], low: datetime, now: datetime, last: date
) -> tuple[Mark, ...]:
    """進捗度が未入力から入ったイベント(作成と退場は除く)の日ごとのタスク名。"""
    days: dict[date, list[str]] = {}
    for item in tasks:
        for event in item.events:
            changes = {c.field: c for c in event.changes}
            progress, status = changes.get("progress"), changes.get("status")
            if progress is None or progress.before is not None or progress.after is None:
                continue
            if status is not None and status.before is None:  # 作成
                continue
            if low <= event.at <= now and event.at.date() <= last:
                days.setdefault(event.at.date(), []).append(item.task.name)
    return tuple(Mark(day, tuple(names)) for day, names in sorted(days.items()))


def compute_burndown(
    tasks: Sequence[BurnTask],
    first: date,
    last: date,
    now: datetime,
    since: datetime | None,
    holidays: dict[date, str],
) -> Burndown:
    """期間(`first`〜`last`)の残りの工数の推移。期間の始めが記録開始より前なら不明。"""
    low = datetime.combine(first, time.min)
    if since is None or low < since:
        return Burndown(first, last, False, since.date() if since else None)
    days = (last - first).days + 1
    today = now.date()
    moments = [low]
    for index in range(days):
        end = datetime.combine(first + timedelta(index + 1), time.min)
        if end <= now:
            moments.append(end)
        else:
            if first + timedelta(index) <= today:  # 今日が期間の中: 日の終わりでなく「今」
                moments.append(now)
            break
    per_task, totals = _by_task(tasks, moments)
    initial, actual = totals[0], tuple(totals[1:])
    planned = planned_curve(
        tasks, [row[0] for row in per_task], first - timedelta(1), last, holidays
    )
    offset = (today - first).days
    today_index = offset if 0 <= offset < days else None
    future: tuple[float, ...] = ()
    fraction = 0.0
    rest = required = None
    recent, recent_days = None, 0
    if today_index is not None:
        fraction = (now - datetime.combine(today, time.min)).total_seconds() / 86400
        tail = planned_curve(tasks, [row[-1] for row in per_task], today, last, holidays)
        future = (actual[-1], *tail)
        rest = actual[-1] - future[-1]
        days_left = count_workdays(today + timedelta(1), last, holidays)
        required = rest / days_left if days_left else None
        recent, recent_days = _recent_pace(tasks, today, now, since, holidays)
    behind = (
        required is not None
        and required > 0
        and recent is not None
        and required > BEHIND_RATIO * recent
    )
    ahead = (
        today_index is not None
        and len(planned) > today_index
        and actual[-1] <= planned[today_index] + 1e-6
    )
    unset = tuple(
        item.task.name
        for item in tasks
        if item.task.kind is TaskKind.NORMAL
        and item.task.status in (Status.RUNNING, Status.PAUSED)
        and item.task.progress is None
    )
    return Burndown(
        first,
        last,
        True,
        since.date(),
        len(tasks),
        initial,
        actual,
        planned,
        future,
        tuple(is_workday(first + timedelta(i), holidays) for i in range(days)),
        today_index,
        fraction,
        _marks(tasks, low, now, last),
        unset,
        rest,
        required,
        recent,
        recent_days,
        behind,
        ahead,
    )
```

```python
        start.date() if start else None,
        end.date() if end else None,
    )


def summarize_burndown(
    project: Project,
    period: Period,
    now: datetime,
    holidays: dict[date, str],
    member: Scope = None,
) -> Burndown:
    """期間中の残りの工数の推移(要望 128 段階 2)。対象は`進んだ分`と同じ集合(予定の期間・締切・実績が期間にかかる
    通常タスク)に、期間中に分解でまとめになったタスク(退場まで数える)を足したもの。参照先は含めない。"""
    index = events_by_task(project)
    schedule = Schedule(project, holidays)
    tasks = [
        _burn_task(entry.task, index, entry.schedule)
        for entry in _entries(project, holidays, member, None)
        if entry.task.kind is TaskKind.NORMAL
        and _touches(entry.task, entry.schedule, period, now)
    ]
    scoped = _names(member)
    for task in project.all_tasks():
        if task.kind is TaskKind.NORMAL or (
            scoped is not None and scoped.isdisjoint(task.assignee_names)
        ):
            continue
        if any(
            c.field == "status" and c.before is not None and c.after is None
            for e in index.get(task.id, ())
            for c in e.changes
        ):
            tasks.append(_burn_task(task, index, schedule))
    return compute_burndown(
        tasks, period.start, period.end, now, project.history_since, holidays
    )
```

```python
"""残りの工数の推移の SVG(要望 128 段階 2)。NiceGUI に依存しない。

横軸は期間の日(位置 i は i 日目の始め = i-1 日目の終わり)、縦軸は残りの工数[h]。
"""

import html
from datetime import timedelta
from math import ceil, floor, log10

from projectapp.burndown import Burndown

WIDTH, HEIGHT = 640, 250
LEFT, RIGHT, TOP, BOTTOM = 46, 14, 30, 36
PLANNED_COLOR = "#8a8f98"
ACTUAL_COLOR = "var(--q-primary, #1976d2)"
FUTURE_COLOR = "var(--q-secondary, #26a69a)"
TODAY_COLOR = "var(--today, #e53935)"


def y_scale(value: float) -> tuple[float, float]:
    """縦軸の上端と目盛りの間隔(1・2・5 × 10^k)。目盛りはおよそ 4 本。"""
    top = max(value, 1.0)
    raw = top / 4
    unit = 10 ** floor(log10(raw))
    step = next(m * unit for m in (1, 2, 5, 10) if m * unit >= raw)
    return float(step * ceil(top / step - 1e-9)), float(step)


def _line(
    points: list[tuple[float, float]], color: str, width: float, dash: str | None
) -> str:
    if len(points) < 2:
        return ""
    coords = " ".join(f"{px:.1f},{py:.1f}" for px, py in points)
    dash_attr = f' stroke-dasharray="{dash}"' if dash else ""
    return (
        f'<polyline fill="none" stroke="{color}" stroke-width="{width}"'
        f'{dash_attr} points="{coords}"/>'
    )


def chart_svg(b: Burndown) -> str:
    days = (b.last - b.first).days + 1
    plot_w, plot_h = WIDTH - LEFT - RIGHT, HEIGHT - TOP - BOTTOM
    ceiling, step = y_scale(max([b.initial, *b.actual, *b.planned, *b.future]))

    def x(position: float) -> float:
        return LEFT + plot_w * position / days

    def y(value: float) -> float:
        return TOP + plot_h * (1 - value / ceiling)

    parts: list[str] = []
    for index, open_ in enumerate(b.workdays):  # 休みの日の帯
        if not open_:
            parts.append(
                f'<rect class="band" x="{x(index):.1f}" y="{TOP}" width="{plot_w / days:.1f}" '
                f'height="{plot_h}" fill="currentColor" opacity="0.06"/>'
            )
    for tick in range(round(ceiling / step) + 1):  # 横線と目盛り
        value = tick * step
        parts.append(
            f'<line x1="{LEFT}" y1="{y(value):.1f}" x2="{WIDTH - RIGHT}" y2="{y(value):.1f}" '
            f'stroke="currentColor" opacity="0.15"/>'
        )
        parts.append(
            f'<text x="{LEFT - 6}" y="{y(value) + 3:.1f}" text-anchor="end" font-size="10" '
            f'fill="currentColor" opacity="0.7">{value:g}</text>'
        )
    label_every = max(1, ceil(days / 10))
    for index in range(0, days, label_every):  # 日付
        day = b.first + timedelta(index)
        parts.append(
            f'<text x="{x(index + 0.5):.1f}" y="{HEIGHT - BOTTOM + 14}" text-anchor="middle" '
            f'font-size="10" fill="currentColor" opacity="0.7">{day:%m/%d}</text>'
        )
    planned = [(0.0, b.initial)] + [(i + 1.0, v) for i, v in enumerate(b.planned)]
    parts.append(_line([(x(p), y(v)) for p, v in planned], PLANNED_COLOR, 2, "2 4"))
    actual = [(0.0, b.initial)]
    for index, value in enumerate(b.actual):
        today = b.today_index == index
        actual.append((index + (b.now_fraction if today else 1.0), value))
    parts.append(_line([(x(p), y(v)) for p, v in actual], ACTUAL_COLOR, 2.5, None))
    if b.today_index is not None and b.future:
        start = b.today_index + b.now_fraction
        future = [(start, b.future[0])] + [
            (b.today_index + 2.0 + j, v) for j, v in enumerate(b.future[1:])
        ]
        parts.append(_line([(x(p), y(v)) for p, v in future], FUTURE_COLOR, 2, "6 4"))
    for index, (position, value) in enumerate(actual[1:]):  # 日ごとの点
        day = b.first + timedelta(index)
        parts.append(
            f'<circle cx="{x(position):.1f}" cy="{y(value):.1f}" r="3" fill="{ACTUAL_COLOR}">'
            f"<title>{day:%m/%d} 残り {value:.1f}h</title></circle>"
        )
    if b.today_index is not None:
        px = x(b.today_index + b.now_fraction)
        parts.append(
            f'<line class="today" x1="{px:.1f}" y1="{TOP}" x2="{px:.1f}" y2="{TOP + plot_h}" '
            f'stroke="{TODAY_COLOR}" stroke-width="1.5"/>'
        )
    base = TOP + plot_h
    for mark in b.marks:  # 進捗度が未入力から入った日
        mx = x((mark.day - b.first).days + 0.5)
        names = html.escape("、".join(mark.names))
        parts.append(
            f'<polygon class="mark" points="{mx - 5:.1f},{base + 9} {mx + 5:.1f},{base + 9} '
            f'{mx:.1f},{base + 2}" fill="{ACTUAL_COLOR}"><title>進捗度を入力({names})</title></polygon>'
        )
    legend = [(PLANNED_COLOR, "2 4", "当初の予定"), (ACTUAL_COLOR, None, "実際")]
    if b.future:
        legend.append((FUTURE_COLOR, "6 4", "今後の予定"))
    for index, (color, dash, text) in enumerate(legend):
        lx = LEFT + index * 120
        dash_attr = f' stroke-dasharray="{dash}"' if dash else ""
        parts.append(
            f'<line x1="{lx}" y1="14" x2="{lx + 24}" y2="14" stroke="{color}" '
            f'stroke-width="2"{dash_attr}/>'
            f'<text x="{lx + 30}" y="18" font-size="11" fill="currentColor">{text}</text>'
        )
    if b.marks:
        parts.append(
            f'<text x="{LEFT + len(legend) * 120}" y="18" font-size="11" fill="currentColor">'
            f"▲ 進捗度を入力</text>"
        )
    return (
        f'<svg viewBox="0 0 {WIDTH} {HEIGHT}" width="100%" role="img" '
        f'aria-label="残りの工数の推移" style="max-width: {WIDTH}px; height: auto">'
        + "".join(parts)
        + "</svg>"
    )
```

```python
    """`残りの工数の推移`のグラフの下の行(要望 128 段階 2)。"""
    if not b.known:
        if b.since is None:
            return [BurnLine("unknown", "履歴はまだありません")]
        return [BurnLine("unknown", f"履歴は {b.since:%m/%d} から(それ以前は不明)")]
    lines: list[BurnLine] = []
    if b.today_index is not None and b.future:
        lines.append(
            BurnLine(
                "end",
                f"期間の終わりの予定の残り {b.future[-1]:.1f}h(今日の実際 {b.actual[-1]:.1f}h)",
            )
        )
    elif b.actual:
        lines.append(
            BurnLine(
                "end",
                f"期間の終わりの実際の残り {b.actual[-1]:.1f}h(予定 {b.planned[-1]:.1f}h)",
            )
        )
    elif b.planned:
        lines.append(
            BurnLine(
                "end",
                f"期間の始めの残り {b.initial:.1f}h(予定どおりなら終わりに {b.planned[-1]:.1f}h)",
            )
        )
    if b.required is not None and b.rest is not None:
        recent = (
            f"直近 {b.recent_days} 稼働日のペース {b.recent:.1f}h/稼働日"
            if b.recent is not None
            else "直近のペースが出ていません"
        )
        lines.append(
            BurnLine(
                "pace",
                f"必要なペース {b.required:.1f}h/稼働日({b.last:%m/%d} までに {b.rest:.1f}h 減らす)・{recent}",
            )
        )
    if b.behind:
        lines.append(BurnLine("behind", "追いついていません"))
    if b.ahead:
        lines.append(BurnLine("ahead", "予定より進んでいます"))
    if b.unset:
        lines.append(
            BurnLine(
                "unset",
                f"進捗度なしの実行中・一時停止 {len(b.unset)} 件(終了まで減りません)",
            )
        )
    return lines
```
